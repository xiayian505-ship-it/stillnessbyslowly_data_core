最簡單呼叫

const art = SlowlyTextGenerator.create({
  maxChars: 4,
  sourceSize: {width: 840, height: 1200},
  renderSVG(text, helpers) {
    // 宿主自己繪製圖案；helpers.escapeXML(text) 可安全寫入 XML
    return `<svg xmlns="http://www.w3.org/2000/svg" width="840" height="1200" viewBox="0 0 840 1200">
       <text x="420" y="600" text-anchor="middle" font-size="180">${helpers.escapeXML(text)}</text>
    </svg>`;
  },
  presets: {
    png: {scale:2},
    cover: {width:700, height:1000, fit:"contain",paddingX:55,paddingY:45}
  },
  filenames: {
    svg: text => `${text}.svg`,
    png: text => `${text}_2x.png`,
    cover: text => `${text}_700x1000.png`
  }
});
art.generateSVG("電閃雷鳴");
art.downloadSVG("電閃雷鳴");
await art.downloadPNG("電閃雷鳴", "png");
await art.downloadCover("電閃雷鳴");

const preview = art.createPreview(document.querySelector("#preview-img"));
preview.set("電閃雷鳴");
preview.clear();
preview.destroy();

art.pngBlob(input, preset)       // { blob, width, height }
art.downloadSVG(input)
art.downloadPNG(input, preset)
art.downloadCover(input)

presets 可以設任意名稱，{scale:2} 表示依宿主 SVG 尺寸等比例放大；或設定固定 width / height 和 fit:"contain"，讓產物保持比例。paddingX / paddingY 為輸出留白。

sourceSize 可為函式，解決星辰版按字數改變 SVG 高度的特例。

PNG 使用軍火庫 SlowlyCanvasPNG.toBlob 與 SlowlyPNGDownload.fromBlob；缺少零件時會明確報錯，不會默默輸出錯誤檔案。生成器不主動上傳或儲存使用者文字或字型。若宿主提供可上傳字型，字型檔授權責任應另在宿主頁面交代。
